# Programs two cues over OSC and leaves the stage dark, then listens to Eos
# while cinq's network cues fire them
probe 3037:slip
send /eos/subscribe 1
wait 3
send /eos/newcmd "Patch#"
wait 2
send /eos/newcmd "Chan 1 Thru 3 Address 1#"
wait 2
send /eos/newcmd "Live#"
wait 2
send /eos/newcmd "Chan 1 At 70#"
wait 1
send /eos/newcmd "Record Cue 1 Time 3#"
wait 2
send /eos/newcmd "Chan 2 At 80#"
wait 1
send /eos/newcmd "Chan 3 At Full#"
wait 1
send /eos/newcmd "Record Cue 2 Time 5#"
wait 2
send /eos/newcmd "Chan 1 Thru 3 At 20#"
wait 1
send /eos/newcmd "Record Cue 3 Time 2#"
wait 2
send /eos/newcmd "Chan 1 Thru 3 At 0#"
wait 3
mark programmed
wait 300
